🔐 Angular 19 Token-Based Authentication (JWT)

This guide walks you through implementing token-based authentication using Angular 19, including how to:

  • Build a login form
  • Call a REST API for login
  • Store and use the token (JWT)
  • Protect routes with AuthGuard

📘 What is Token-Based Authentication?

A server issues a token (usually a JWT) after verifying user credentials. The frontend app stores this token and includes it in future API requests.

🔁 Basic Flow

  1. User logs in → API returns token
  2. Token is stored in localStorage
  3. Every request sends token in headers
  4. Backend checks token → grants or denies access

🔧 1. AuthService – Login, Store Token

      
        import { Injectable } from '@angular/core';
        import { HttpClient } from '@angular/common/http';
        import { Observable, tap } from 'rxjs';

        @Injectable({ providedIn: 'root' })
        export class AuthService {
        private loginUrl = 'https://example.com/api/login'; // Change to your API

        constructor(private http: HttpClient) {}

        login(data: { email: string; password: string }): Observable<any> {
          return this.http.post(this.loginUrl, data).pipe(
          tap((res: any) => localStorage.setItem('token', res.token))
          );
        }

        logout(): void {
          localStorage.removeItem('token');
        }

        getToken(): string | null {
          return localStorage.getItem('token');
        }

        isLoggedIn(): boolean {
          return !!this.getToken();
        }
      }

🧩 2. Login Component

      
        import { Component } from '@angular/core';
        import { FormsModule } from '@angular/forms';
        import { CommonModule } from '@angular/common';
        import { AuthService } from '../services/auth.service';
        import { HttpClientModule } from '@angular/common/http';

        @Component({
          selector: 'app-login',
          standalone: true,
          imports: [CommonModule, FormsModule, HttpClientModule],
          template: `
          <h3>Login</h3>
          <form (ngSubmit)="onSubmit()" #f="ngForm">
          <input name="email" [(ngModel)]="email" required placeholder="Email"><br><br>
          <input name="password" [(ngModel)]="password" type="password" required placeholder="Password"><br><br>
          <button type="submit">Login</button>
          </form>
        <p *ngIf="error">

          {{ error }}
</p>
          `
          })
        export class LoginComponent {
            email = '';
            password = '';
            error = '';

            constructor(private auth: AuthService) {}

            onSubmit() {
              this.auth.login({ email: this.email, password: this.password }).subscribe({
              next: () => alert('Login successful'),
                error: () => this.error = 'Login failed'
              });
            }
        }

🛡️ 3. AuthGuard – Protect Routes

      
        import { Injectable } from '@angular/core';
        import { CanActivateFn } from '@angular/router';
        import { AuthService } from '../services/auth.service';

        export const authGuard: CanActivateFn = () => {
          return inject(AuthService).isLoggedIn();
        };

📎 4. Token Interceptor – Attach Token

      
        import { Injectable } from '@angular/core';
        import { HttpInterceptor, HttpRequest, HttpHandler } from '@angular/common/http';
        import { AuthService } from '../services/auth.service';
        import { Observable } from 'rxjs';

        @Injectable()
        export class TokenInterceptor implements HttpInterceptor {
        constructor(private auth: AuthService) {}

        intercept(req: HttpRequest<any>, next: HttpHandler): Observable<any> {
          const token = this.auth.getToken();
          if (token) {
          req = req.clone({
          headers: req.headers.set('Authorization', 'Bearer ' + token)
          });
        }
          return next.handle(req);
          }
        }

🔁 5. Provide Interceptor in Main

      
    import { bootstrapApplication } from '@angular/platform-browser';
    import { AppComponent } from './app/app.component';
    import { provideHttpClient, withInterceptors } from '@angular/common/http';
    import { TokenInterceptor } from './app/interceptors/token.interceptor';

    bootstrapApplication(AppComponent, {
      providers: [provideHttpClient(withInterceptors([TokenInterceptor]))]
    });

✅ Summary

  • Login: AuthService sends credentials and stores token
  • Storage: Token saved in localStorage
  • Interceptor: Automatically attaches token to requests
  • Guard: Prevents access to routes without token

Token-based authentication keeps Angular apps stateless, secure, and simple!
